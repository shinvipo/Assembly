
INCLUDE Irvine32.inc

.data
		fibo		word	?
		count		dword	5
		sz			equ		type fibo
.code
main	proc
		mov ecx	, count
		mov esi, offset fibo
		mov eax, [esi]
		mov eax,1
		mov [esi], eax
		add esi, sz
		mov eax, [esi]
		mov eax,1
		mov [esi], eax
		add esi, sz
	l1:
		mov	ax, [esi - sz]
		mov	bx, [esi - sz * 2]
		
		add ax, bx

		mov word ptr [esi], ax
		add esi, sz

		loop l1

		call DumpRegs
		exit
main endp
END main
