
INCLUDE Irvine32.inc

.data
		source		byte		"this is source string!",0
		des		byte		SIZEOF source DUP ('#')	
.code
main	proc
		mov		eax,0
		mov		ebx,0
		mov		esi,0
		mov		edi,lengthof source -1
		mov		ecx,lengthof source

	l1:
		mov		al,source[edi]
		mov		des[esi],al

		inc		esi
		dec		edi

		loop l1
		call DumpRegs
		exit
main endp
END main
