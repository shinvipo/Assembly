
INCLUDE Irvine32.inc

.data
		arr		word	1,2,3,4,5
		len		equ		lengthof arr
		tp		equ		type	arr
		sz		equ		sizeof	arr
.code
main	proc
		mov			eax,0
		mov			ebx,0
		mov			esi,0
		mov			edi,len*2 -2
		mov			ecx, len/2
	l1:
		mov			ax, arr[esi]
		mov			bx,	arr[edi]

		xchg		eax,ebx

		mov			word ptr arr[esi],ax
		mov			word ptr arr[edi],bx

		add			esi, tp
		sub			edi, tp

		loop l1

		call DumpRegs
		exit
main endp
END main
